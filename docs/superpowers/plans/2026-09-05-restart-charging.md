# Restart Charging Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Read the car's named charging fault over BLE, show it like Tesla's ChargingAlerts row, and offer a `Restart Charging` action that performs an alert-driven software re-plug (unlatch → dwell → start → verify, with an amps step-down) from anywhere the phone can reach the car.

**Architecture:** Three pure modules carry the logic — the alert table + parser (`chargingAlerts.ts`), the telemetry parse of the new ChargeState/AlertState fields (`telemetry.ts`), and the restart state machine with injected IO (`chargeRestart.ts`). `useCarLink` binds the machine to the gateway (commands + scoped reads + wake) and exposes status; two screens render one `ChargingAlertCard`.

**Tech Stack:** Expo SDK 56 / React Native, TypeScript strict, protobufjs static bindings (`npm run build:teslaproto`), node:test via tsx (`npm test`), iOS JS-only deploy `bash scripts/godot-ios/deploy-js.sh`.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-09-05-restart-charging-design.md`; research: `docs/superpowers/research/tesla-charging-alerts-FINDINGS.md`.
- Pure `src/ble/*` modules import NO React Native; tests run under plain node (`npm test`).
- Every telemetry key must be cached or excused (`carLinkCacheCoverage.test.ts`) and every cached key restored (`rehydrateCoverage.test.ts`).
- `Start Charging` / `Unlock Charge Port` keep their parity behaviour; the restart is a NEW button.
- Macro defaults: max 3 attempts; dwell 15 s / 45 s / 120 s; unlatch wait 15 s (poll 2.5 s); verify 90 s (poll 3 s); start grace 20 s; amps step `max(5, floor(amps × 0.7))`.
- Families `equipment` and `dc` get one attempt then `needs-charger`.
- Never `expo prebuild`. No commits unless Ivan asks; leave the tree for review.

```ts
// chargingAlerts — the car's own name for why charging stopped or slowed, with the
// words Tesla's app prints under it.
//
// Source: VehicleData.alert_state (GetVehicleData.getAlertState = 20 → VehicleData 25),
// `repeated ChargingAlert { alert_name: ChargingAlertName; alert_type: ChargingAlertType }`.
// The car serves it locally (QtCarServer alert_state handler); the official app has the
// UI (VehicleChargingAlertsRow) but never requests it over BLE. We do. Full recovery in
// docs/superpowers/research/tesla-charging-alerts-FINDINGS.md.
//
// PURE — no RN imports; the table is data (chargingAlertTable.ts, generated).

import type { ChargingAlertFamily, ChargingAlertInfo, ChargingAlertKind } from '../types/vehicleTypes';
import { CHARGING_ALERT_TABLE, type ChargingAlertRow } from './chargingAlertTable';

const BY_N = new Map<number, ChargingAlertRow>(CHARGING_ALERT_TABLE.map((r) => [r.n, r]));
const BY_NAME = new Map<string, ChargingAlertRow>(CHARGING_ALERT_TABLE.map((r) => [r.name, r]));

// alertFamily — which lever can clear it, keyed on Tesla's own remedy wording. FIRST
// matching rule wins, so the order is the policy:
//   info      a reduced-rate / heating notice: the car is still charging, nothing to restart
//   replug    Tesla says unplug / disconnect / reinsert / try again — the software re-plug
//   voltage-drop  "remove extension cords": V = I·R, so a lower current is the remedy
//   thermal   something is hot: dwell (cool down) then retry, lower current second time
//   power     the supply went away: retry, it may be back
//   equipment wiring / GFCI / service / configuration: the car cannot clear these
//   dc        Supercharger / DC adapter / station faults: not a home-AC problem
export function alertFamily(row: { title: string; subtitle: string }): ChargingAlertFamily {
  const text = `${row.title} ${row.subtitle}`.toLowerCase();
  if (/rate reduced|charging slowed|is heating|rate adjusted|rate will increase/.test(text)) return 'info';
  if (/unplug|disconnect|insert charge handle|plug adapter fully|try again/.test(text)) return 'replug';
  if (/voltage drop|extension cords|line impedance/.test(text)) return 'voltage-drop';
  if (/temperature|too hot|connection hot|cool/.test(text)) return 'thermal';
  if (/power lost|not providing power|not powered|grid|powerwall|ac power system/.test(text)) return 'power';
  if (/service|inspect|wiring|grounding|gfci|primary|rotary|configuration|switch setting|current ratings|wall connector issue/.test(text)) {
    return 'equipment';
  }
  if (/supercharg|dc fast|stall|station|chademo|adapter|v2g|handle incompatible|arc flash/.test(text)) return 'dc';
  return 'other';
}

function kindOf(v: unknown): ChargingAlertKind {
  if (v === 1 || v === 'WARNING') return 'warning';
  if (v === 2 || v === 'NOTIFICATION') return 'notification';
  return 'unknown';
}

// chargingAlertInfo — look one alert up by wire number OR enum name. An unknown number
// is still returned (as ALERT_<n>) so a newer firmware's fault is shown rather than
// swallowed; INVALID_ALERT_NAME (0) and junk return null.
export function chargingAlertInfo(nameOrNumber: unknown, kind: ChargingAlertKind = 'unknown'): ChargingAlertInfo | null {
  let n: number | null = null;
  let row: ChargingAlertRow | undefined;
  if (typeof nameOrNumber === 'number' && Number.isFinite(nameOrNumber)) {
    n = nameOrNumber;
    row = BY_N.get(n);
  } else if (typeof nameOrNumber === 'string') {
    row = BY_NAME.get(nameOrNumber);
    if (row) n = row.n;
    else if (/^\d+$/.test(nameOrNumber)) {
      n = Number(nameOrNumber);
      row = BY_N.get(n);
    }
  }
  if (n === null || n === 0) return null;
  if (!row) {
    return { n, name: `ALERT_${n}`, title: `Charging alert ${n}`, subtitle: 'Unknown alert — check the car screen', family: 'other', kind };
  }
  return { n: row.n, name: row.name, title: row.title, subtitle: row.subtitle, family: alertFamily(row), kind };
}

// parseChargingAlerts — a decoded CarServer.AlertState (or a toObject of one) → infos.
// Tolerant of both numeric and string enums, drops junk, collapses duplicates.
export function parseChargingAlerts(alertState: unknown): ChargingAlertInfo[] {
  if (!alertState || typeof alertState !== 'object') return [];
  const list = (alertState as { chargingAlerts?: unknown }).chargingAlerts;
  if (!Array.isArray(list)) return [];
  const out: ChargingAlertInfo[] = [];
  const seen = new Set<string>();
  for (const item of list) {
    if (!item || typeof item !== 'object') continue;
    const a = item as { alertName?: unknown; alertType?: unknown };
    const info = chargingAlertInfo(a.alertName, kindOf(a.alertType));
    if (!info || seen.has(info.name)) continue;
    seen.add(info.name);
    out.push(info);
  }
  return out;
}

// primaryChargingAlert — the one the card leads with and the restart plans around: the
// first actionable (non-info) alert, warnings before notifications. null = nothing to fix.
export function primaryChargingAlert(alerts: readonly ChargingAlertInfo[]): ChargingAlertInfo | null {
  const actionable = alerts.filter((a) => a.family !== 'info');
  return actionable.find((a) => a.kind === 'warning') ?? actionable[0] ?? null;
}
```

//
// What a hand does at the car: unlock the port, pull the cable, push it back. What this
// does from the couch: unlatch (VCSEC), wait, ask for charging again (which re-engages the
// latch), and WATCH the charge state long enough to know it stuck — the app's plain Start
// was accepted by the car every time and still reverted after ~30 s, so acceptance proves
// nothing; only a sustained Charging read does.
//
// It plans around the car's own fault name (VehicleData.alert_state → chargingAlerts):
//   voltage-drop  lower the current from the first attempt (V = I·R on the house wiring)
//   thermal/other/none  plain cycle first, lower current from the second
//   replug/power  plain cycles, lower current only as a last resort
//   equipment/dc  one plain attempt, then stop: the car cannot clear those
//

type Family = ChargingAlertFamily | 'none';

// Attempts allowed per family; absent = the option's maxAttempts.
const MAX_ATTEMPTS_BY_FAMILY: Partial<Record<Family, number>> = { equipment: 1, dc: 1 };
// First attempt (1-based) on which the current is stepped down.
const LOWER_AMPS_FROM: Record<Family, number> = {
  'voltage-drop': 1,
  thermal: 2,
  other: 2,
  none: 2,
  replug: 3,
  power: 3,
  info: Number.POSITIVE_INFINITY,
  equipment: Number.POSITIVE_INFINITY,
  dc: Number.POSITIVE_INFINITY,
};

export function planAttempt(
  attempt: number,
  family: Family,
  opts: Pick<ChargeRestartOptions, 'maxAttempts' | 'dwellMs'> = {},
): { dwellMs: number; lowerAmps: boolean; maxAttempts: number } {
  const maxAttempts = Math.min(opts.maxAttempts ?? CHARGE_RESTART_DEFAULTS.maxAttempts, MAX_ATTEMPTS_BY_FAMILY[family] ?? Number.POSITIVE_INFINITY);
  const dwells = opts.dwellMs ?? CHARGE_RESTART_DEFAULTS.dwellMs;
  const dwellMs = dwells[Math.min(Math.max(attempt, 1), dwells.length) - 1];
  return { dwellMs, lowerAmps: attempt >= LOWER_AMPS_FROM[family], maxAttempts };
}

export async function restartCharging(io: ChargeRestartIo, o: ChargeRestartOptions = {}): Promise<ChargeRestartResult> {
  const opts = { ...CHARGE_RESTART_DEFAULTS, ...o };
  const signal = o.signal;
  const trace: string[] = [];
  const say = (step: string, data?: Record<string, string | number | boolean>) => {
    trace.push(data ? `${step} ${JSON.stringify(data)}` : step);
    io.log(step, data);
  };
  const aborted = () => signal?.aborted === true;
  const progress = (attempt: number, maxAttempts: number, phase: ChargeRestartPhase, untilMs: number | null = null) =>
    io.onProgress?.({ attempt, maxAttempts, phase, untilMs });
  const safeAlerts = async (): Promise<ChargingAlertInfo[]> => {
    try {
      return await io.readAlerts();
    } catch (e) {
      say('alerts read failed', { error: e instanceof Error ? e.message : String(e) });
      return [];
    }
  };

  let attempt = 0;
  let alertsBefore: ChargingAlertInfo[] = [];
  let alertsAfter: ChargingAlertInfo[] = [];
  let ampsChangedTo: number | null = null;
  let last: ChargeRead | null = null;
  const done = (ok: boolean, reason: ChargeRestartReason): ChargeRestartResult => {
    progress(attempt, opts.maxAttempts, 'done');
    say('result', { ok, reason, attempts: attempt });
    return { ok, reason, attempts: attempt, alertsBefore, alertsAfter, ampsChangedTo, finalState: last, trace };
  };

  progress(0, opts.maxAttempts, 'reading');
  alertsBefore = await safeAlerts();
  try {
    last = await io.readCharge();
  } catch (e) {
    say('charge read failed', { error: e instanceof Error ? e.message : String(e) });
    return done(false, 'read-failed');
  }
  say('before', {
    state: last.chargingState ?? '-',
    latch: last.chargePortLatch ?? '-',
    cableUnlatched: String(last.chargeCableUnlatched),
    amps: last.chargingAmps ?? -1,
    alerts: names(alertsBefore),
  });
  if (isCharging(last)) return done(true, 'charging');
  if (!isPlugged(last)) return done(false, 'no-cable');

  let family = familyOf(alertsBefore);
  let lastReason: ChargeRestartReason = 'never-started';

  for (;;) {
    const plan = planAttempt(attempt + 1, family, opts);
    if (attempt >= plan.maxAttempts) {
      return done(false, plan.maxAttempts < opts.maxAttempts ? 'needs-charger' : lastReason);
    }
    attempt += 1;
    if (aborted()) return done(false, 'aborted');
    say('attempt', { attempt, family, dwellMs: plan.dwellMs, lowerAmps: plan.lowerAmps });

    // 1. Unlatch — the same VCSEC action as the Unlock Charge Port button.
    progress(attempt, plan.maxAttempts, 'unlatching', io.now() + opts.unlatchTimeoutMs);
    const un = await io.runCommand({ type: 'openChargePort' });
    say('unlatch', { ok: un.ok, message: un.message ?? '' });
    if (aborted()) return done(false, 'aborted');
    const unlatchDeadline = io.now() + opts.unlatchTimeoutMs;
    let unlatched = false;
    while (io.now() < unlatchDeadline) {
      await io.sleep(opts.unlatchPollMs, signal);
      if (aborted()) return done(false, 'aborted');
      try {
        last = await io.readCharge();
      } catch {
        continue;
      }
      if (last.chargePortLatch === 'disengaged' || last.chargeCableUnlatched === true) {
        unlatched = true;
        break;
      }
    }
    say('unlatched', { seen: unlatched, latch: last?.chargePortLatch ?? '-', cableUnlatched: String(last?.chargeCableUnlatched) });

    // 2. Dwell — the pause a hand takes between pulling and pushing, longer each attempt.
    progress(attempt, plan.maxAttempts, 'waiting', io.now() + plan.dwellMs);
    await io.sleep(plan.dwellMs, signal);
    if (aborted()) return done(false, 'aborted');

    // 3. Current step-down when the plan says so. Never raised back; the result reports it.
    const amps = last?.chargingAmps ?? null;
    if (plan.lowerAmps && amps !== null && amps > opts.minAmps) {
      const target = Math.max(opts.minAmps, Math.floor(amps * opts.ampsFactor));
      if (target < amps) {
        progress(attempt, plan.maxAttempts, 'lowering-amps');
        const r = await io.runCommand({ type: 'setChargingAmps', amps: target });
        say('amps', { from: amps, to: target, ok: r.ok, message: r.message ?? '' });
        if (r.ok) {
          ampsChangedTo = target;
          if (last) last = { ...last, chargingAmps: target };
        }
        if (aborted()) return done(false, 'aborted');
      }
    }

    // 4. Ask for charging — this is what re-engages the latch.
    progress(attempt, plan.maxAttempts, 'starting');
    const st = await io.runCommand({ type: 'chargeStart' });
    say('start', { ok: st.ok, message: st.message ?? '' });
    let verdict: 'charging' | 'complete' | 'stopped' | 'never' | 'rejected' = st.ok ? 'never' : 'rejected';

    // 5. Verify — watch it past the window the plain Start used to revert in.
    if (st.ok) {
      const t0 = io.now();
      progress(attempt, plan.maxAttempts, 'verifying', t0 + opts.verifyMs);
      let sawLive = false;
      while (io.now() - t0 < opts.verifyMs) {
        await io.sleep(opts.verifyPollMs, signal);
        if (aborted()) return done(false, 'aborted');
        try {
          last = await io.readCharge();
        } catch {
          continue;
        }
        const s = last.chargingState;
        if (s === 'charging' || s === 'starting') sawLive = true;
        if (s === 'complete') {
          verdict = 'complete';
          break;
        }
        if ((s === 'stopped' || s === 'nopower') && (sawLive || io.now() - t0 >= opts.startGraceMs)) {
          verdict = 'stopped';
          break;
        }
      }
      if (verdict === 'never' && isCharging(last)) verdict = 'charging';
      say('verify', {
        verdict,
        state: last?.chargingState ?? '-',
        currentA: last?.chargerActualCurrentA ?? -1,
        latch: last?.chargePortLatch ?? '-',
        ms: io.now() - t0,
      });
      if (verdict === 'charging' || verdict === 'complete') return done(true, verdict);
    }

    // Failed attempt: ask the car why, re-plan around the newest alert.
    alertsAfter = await safeAlerts();
    say('after', { alerts: names(alertsAfter) });
    const next = familyOf(alertsAfter);
    if (next !== 'none') family = next;
    lastReason = verdict === 'rejected' ? 'start-rejected' : verdict === 'stopped' ? 'stopped-again' : 'never-started';
  }
}

export function chargeRestartToast(r: ChargeRestartResult): { title: string; body?: string } {
